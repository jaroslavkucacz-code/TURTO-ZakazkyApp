def _text_path(value) -> str:
    try:return os.path.abspath(os.fspath(value))
    except Exception:return str(value)


def _vbs_quote(value: str) -> str:
    return str(value).replace('"','""')

def ensure_windows_shortcuts(root) -> None:
    if sys.platform!='win32':return
    root_s=_text_path(root)
    native=bool(getattr(sys,'frozen',False))
    if native:
        target=_text_path(sys.executable)
        args=''
        icon=target+',0'
    else:
        exe=Path(sys.executable)
        if exe.name.lower()=='python.exe' and exe.with_name('pythonw.exe').exists():exe=exe.with_name('pythonw.exe')
        target=_text_path(exe)
        app=os.path.join(root_s,'app.pyw')
        args='"'+app+'"'
        icon=os.path.join(root_s,'assets','app_icon.ico')+',0'
    if not os.path.exists(target):return

    lines=[
        'Set shell = CreateObject("WScript.Shell")',
        'desktop = shell.SpecialFolders("Desktop")',
        'programs = shell.SpecialFolders("Programs")',
        f'target = "{_vbs_quote(target)}"',
        f'args = "{_vbs_quote(args)}"',
        f'root = "{_vbs_quote(root_s)}"',
        f'icon = "{_vbs_quote(icon)}"',
        'For Each linkPath In Array(desktop & "\\TURTO - Mesicni prehledy.lnk", programs & "\\TURTO - Mesicni prehledy.lnk", root & "\\TURTO - Mesicni prehledy.lnk")',
        '  Set sc = shell.CreateShortcut(linkPath)',
        '  sc.TargetPath = target',
        '  sc.Arguments = args',
        '  sc.WorkingDirectory = root',
        '  sc.IconLocation = icon',
        '  sc.Description = "TURTO - Mesicni prehledy"',
        '  sc.Save',
        'Next',
    ]
    tmp=None
    try:
        fd,name=tempfile.mkstemp(prefix='turto_shortcut_',suffix='.vbs');os.close(fd);tmp=Path(name)
        tmp.write_text('\r\n'.join(lines)+'\r\n',encoding='utf-8-sig')
        subprocess.run(['cscript.exe','//nologo',str(tmp)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,timeout=8,check=False,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
        try:subprocess.run(['ie4uinit.exe','-show'],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,timeout=5,check=False,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
        except Exception:pass
    except Exception:
        pass
    finally:
        if tmp:
            try:tmp.unlink(missing_ok=True)
            except Exception:pass
